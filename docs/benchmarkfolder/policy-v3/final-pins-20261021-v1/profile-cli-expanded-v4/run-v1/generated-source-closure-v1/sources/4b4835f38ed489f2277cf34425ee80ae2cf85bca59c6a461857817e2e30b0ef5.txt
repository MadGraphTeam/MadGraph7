      SUBROUTINE ML5_0_COEF_CONSTRUCTION_1(P,NHEL,H,IC)
C     
C     Modules
C     
      USE ML5_0_POLYNOMIAL_CONSTANTS
      USE ALOHA_OBJECT
C     
      IMPLICIT NONE
C     
C     CONSTANTS
C     
      INTEGER    NEXTERNAL
      PARAMETER (NEXTERNAL=3)
      INTEGER    NCOMB
      PARAMETER (NCOMB=12)
      INTEGER NBORNAMPS
      PARAMETER (NBORNAMPS=1)
      INTEGER    NLOOPS, NLOOPGROUPS, NCTAMPS
      PARAMETER (NLOOPS=6, NLOOPGROUPS=6, NCTAMPS=3)
      INTEGER    NLOOPAMPS
      PARAMETER (NLOOPAMPS=9)
      INTEGER    NWAVEFUNCS,NLOOPWAVEFUNCS
      PARAMETER (NWAVEFUNCS=3,NLOOPWAVEFUNCS=17)
      REAL*8     ZERO
      PARAMETER (ZERO=0D0)
      REAL*16     MP__ZERO
      PARAMETER (MP__ZERO=0.0E0_16)
C     These are constants related to the split orders
      INTEGER    NSO, NSQUAREDSO, NAMPSO
      PARAMETER (NSO=1, NSQUAREDSO=1, NAMPSO=2)
C     
C     ARGUMENTS
C     
      REAL*8 P(0:3,NEXTERNAL)
      INTEGER NHEL(NEXTERNAL), IC(NEXTERNAL)
      INTEGER H
C     
C     LOCAL VARIABLES
C     
      INTEGER I,J,K
      INTEGER FLAVOR(NEXTERNAL)
      DATA FLAVOR /NEXTERNAL*1/
      COMPLEX*16 COEFS(MAXLWFSIZE,0:VERTEXMAXCOEFS-1,MAXLWFSIZE)

      LOGICAL DUMMYFALSE
      DATA DUMMYFALSE/.FALSE./
C     
C     GLOBAL VARIABLES
C     

      INCLUDE 'coupl.inc'
      INCLUDE 'mp_coupl.inc'

      INTEGER HELOFFSET
      INTEGER GOODHEL(NCOMB)
      LOGICAL GOODAMP(NSQUAREDSO,NLOOPGROUPS)
      COMMON/ML5_0_FILTERS/GOODAMP,GOODHEL,HELOFFSET

      LOGICAL CHECKPHASE
      LOGICAL HELDOUBLECHECKED
      COMMON/ML5_0_INIT/CHECKPHASE, HELDOUBLECHECKED

      INTEGER SQSO_TARGET
      COMMON/ML5_0_SOCHOICE/SQSO_TARGET

      LOGICAL UVCT_REQ_SO_DONE,MP_UVCT_REQ_SO_DONE,CT_REQ_SO_DONE
     $ ,MP_CT_REQ_SO_DONE,LOOP_REQ_SO_DONE,MP_LOOP_REQ_SO_DONE
     $ ,CTCALL_REQ_SO_DONE,FILTER_SO
      COMMON/ML5_0_SO_REQS/UVCT_REQ_SO_DONE,MP_UVCT_REQ_SO_DONE
     $ ,CT_REQ_SO_DONE,MP_CT_REQ_SO_DONE,LOOP_REQ_SO_DONE
     $ ,MP_LOOP_REQ_SO_DONE,CTCALL_REQ_SO_DONE,FILTER_SO

      INTEGER I_SO
      COMMON/ML5_0_I_SO/I_SO
      INTEGER I_LIB
      COMMON/ML5_0_I_LIB/I_LIB

      COMPLEX*16 AMP(NBORNAMPS)
      COMMON/ML5_0_AMPS/AMP
      TYPE(ALOHA) W(NWAVEFUNCS)
      COMMON/ML5_0_W/W
      COMPLEX*16 WL(MAXLWFSIZE,0:LOOPMAXCOEFS-1,MAXLWFSIZE,
     $ -1:NLOOPWAVEFUNCS)
      COMPLEX*16 PL(0:3,-1:NLOOPWAVEFUNCS)
      COMMON/ML5_0_WL/WL,PL

      COMPLEX*16 AMPL(3,NCTAMPS)
      COMMON/ML5_0_AMPL/AMPL

C     
C     ----------
C     BEGIN CODE
C     ----------

C     The target squared split order contribution is already reached
C      if true.
      IF (FILTER_SO.AND.LOOP_REQ_SO_DONE) THEN
        GOTO 1001
      ENDIF

C     Coefficient construction for loop diagram with ID 2
      CALL FFV1L2P0_3(PL(0,0),W(1),GC_2,ZERO,ZERO,PL(0,1),COEFS)
      CALL ML5_0_UPDATE_WL_0_0(WL(1,0,1,0),4,COEFS,4,4,WL(1,0,1,1))
      CALL FFV1L3_1(PL(0,1),W(2),GC_1,ZERO,ZERO,PL(0,2),COEFS)
      CALL ML5_0_UPDATE_WL_0_1(WL(1,0,1,1),4,COEFS,4,4,WL(1,0,1,2))
      CALL FFV2L2_1(PL(0,2),W(3),GC_124,ZERO,ZERO,PL(0,3),COEFS)
      CALL ML5_0_UPDATE_WL_1_1(WL(1,0,1,2),4,COEFS,4,4,WL(1,0,1,3))
      CALL ML5_0_CREATE_LOOP_COEFS(WL(1,0,1,3),2,4,1,1,1,4,H)
C     Coefficient construction for loop diagram with ID 3
      CALL FFV1L3_2(PL(0,0),W(1),GC_2,ZERO,ZERO,PL(0,4),COEFS)
      CALL ML5_0_UPDATE_WL_0_1(WL(1,0,1,0),4,COEFS,4,4,WL(1,0,1,4))
      CALL FFV2L1P0_3(PL(0,4),W(2),GC_124,MDL_MW,MDL_WW,PL(0,5),COEFS)
      CALL ML5_0_UPDATE_WL_1_0(WL(1,0,1,4),4,COEFS,4,4,WL(1,0,1,5))
      CALL VVV1L2P0_1(PL(0,5),W(3),GC_4,ZERO,ZERO,PL(0,6),COEFS)
      CALL ML5_0_UPDATE_WL_1_1(WL(1,0,1,5),4,COEFS,4,4,WL(1,0,1,6))
      CALL ML5_0_CREATE_LOOP_COEFS(WL(1,0,1,6),2,4,2,1,1,5,H)
C     Coefficient construction for loop diagram with ID 4
      CALL FFV2L3_2(PL(0,0),W(1),GC_124,ZERO,ZERO,PL(0,7),COEFS)
      CALL ML5_0_UPDATE_WL_0_1(WL(1,0,1,0),4,COEFS,4,4,WL(1,0,1,7))
      CALL FFV1L1P0_3(PL(0,7),W(2),GC_1,ZERO,ZERO,PL(0,8),COEFS)
      CALL ML5_0_UPDATE_WL_1_0(WL(1,0,1,7),4,COEFS,4,4,WL(1,0,1,8))
      CALL VVV1L1P0_2(PL(0,8),W(3),GC_4,MDL_MW,MDL_WW,PL(0,9),COEFS)
      CALL ML5_0_UPDATE_WL_1_1(WL(1,0,1,8),4,COEFS,4,4,WL(1,0,1,9))
      CALL ML5_0_CREATE_LOOP_COEFS(WL(1,0,1,9),2,4,3,1,1,6,H)
C     Coefficient construction for loop diagram with ID 5
      CALL FFV2_5L2P0_3(PL(0,0),W(1),GC_69,GC_76,MDL_MZ,MDL_WZ,PL(0,10)
     $ ,COEFS)
      CALL ML5_0_UPDATE_WL_0_0(WL(1,0,1,0),4,COEFS,4,4,WL(1,0,1,10))
      CALL FFV2_3L3_1(PL(0,10),W(2),GC_68,GC_76,ZERO,ZERO,PL(0,11)
     $ ,COEFS)
      CALL ML5_0_UPDATE_WL_0_1(WL(1,0,1,10),4,COEFS,4,4,WL(1,0,1,11))
      CALL FFV2L2_1(PL(0,11),W(3),GC_124,ZERO,ZERO,PL(0,12),COEFS)
      CALL ML5_0_UPDATE_WL_1_1(WL(1,0,1,11),4,COEFS,4,4,WL(1,0,1,12))
      CALL ML5_0_CREATE_LOOP_COEFS(WL(1,0,1,12),2,4,4,1,1,7,H)
C     Coefficient construction for loop diagram with ID 6
      CALL FFV2_5L3_2(PL(0,0),W(1),GC_69,GC_76,ZERO,ZERO,PL(0,13)
     $ ,COEFS)
      CALL ML5_0_UPDATE_WL_0_1(WL(1,0,1,0),4,COEFS,4,4,WL(1,0,1,13))
      CALL FFV2L1P0_3(PL(0,13),W(2),GC_124,MDL_MW,MDL_WW,PL(0,14)
     $ ,COEFS)
      CALL ML5_0_UPDATE_WL_1_0(WL(1,0,1,13),4,COEFS,4,4,WL(1,0,1,14))
      CALL VVV1L1P0_3(PL(0,14),W(3),GC_71,MDL_MZ,MDL_WZ,PL(0,15),COEFS)
      CALL ML5_0_UPDATE_WL_1_1(WL(1,0,1,14),4,COEFS,4,4,WL(1,0,1,15))
      CALL ML5_0_CREATE_LOOP_COEFS(WL(1,0,1,15),2,4,5,1,1,8,H)
C     Coefficient construction for loop diagram with ID 7
      CALL FFV2_3L1P0_3(PL(0,7),W(2),GC_68,GC_76,MDL_MZ,MDL_WZ,PL(0,16)
     $ ,COEFS)
      CALL ML5_0_UPDATE_WL_1_0(WL(1,0,1,7),4,COEFS,4,4,WL(1,0,1,16))
      CALL VVV1L3P0_1(PL(0,16),W(3),GC_71,MDL_MW,MDL_WW,PL(0,17),COEFS)
      CALL ML5_0_UPDATE_WL_1_1(WL(1,0,1,16),4,COEFS,4,4,WL(1,0,1,17))
      CALL ML5_0_CREATE_LOOP_COEFS(WL(1,0,1,17),2,4,6,1,1,9,H)
C     At this point, all loop coefficients needed for (QED=4), i.e. of
C      split order ID=1, are computed.
      IF(FILTER_SO.AND.SQSO_TARGET.EQ.1) GOTO 4000

      GOTO 1001
 4000 CONTINUE
      LOOP_REQ_SO_DONE=.TRUE.
 1001 CONTINUE
      END
