      SUBROUTINE ML5_0_HELAS_CALLS_UVCT_1(P,NHEL,H,IC)
C     
C     Modules
C     
      USE ML5_0_POLYNOMIAL_CONSTANTS
      USE ALOHA_OBJECT
C     
      IMPLICIT NONE
C     
C     CONSTANTS
C     
      INTEGER    NEXTERNAL
      PARAMETER (NEXTERNAL=5)
      INTEGER    NCOMB
      PARAMETER (NCOMB=48)
      INTEGER NBORNAMPS
      PARAMETER (NBORNAMPS=8)
      INTEGER    NLOOPS, NLOOPGROUPS, NCTAMPS
      PARAMETER (NLOOPS=224, NLOOPGROUPS=116, NCTAMPS=278)
      INTEGER    NLOOPAMPS
      PARAMETER (NLOOPAMPS=502)
      INTEGER    NWAVEFUNCS,NLOOPWAVEFUNCS
      PARAMETER (NWAVEFUNCS=35,NLOOPWAVEFUNCS=388)
      REAL*8     ZERO
      PARAMETER (ZERO=0D0)
      REAL*16     MP__ZERO
      PARAMETER (MP__ZERO=0.0E0_16)
C     These are constants related to the split orders
      INTEGER    NSO, NSQUAREDSO, NAMPSO
      PARAMETER (NSO=1, NSQUAREDSO=1, NAMPSO=2)
C     
C     ARGUMENTS
C     
      REAL*8 P(0:3,NEXTERNAL)
      INTEGER NHEL(NEXTERNAL), IC(NEXTERNAL)
      INTEGER H
C     
C     LOCAL VARIABLES
C     
      INTEGER I,J,K
      INTEGER FLAVOR(NEXTERNAL)
      DATA FLAVOR /NEXTERNAL*1/
      COMPLEX*16 COEFS(MAXLWFSIZE,0:VERTEXMAXCOEFS-1,MAXLWFSIZE)

      LOGICAL DUMMYFALSE
      DATA DUMMYFALSE/.FALSE./
C     
C     GLOBAL VARIABLES
C     

      INCLUDE 'coupl.inc'
      INCLUDE 'mp_coupl.inc'

      INTEGER HELOFFSET
      INTEGER GOODHEL(NCOMB)
      LOGICAL GOODAMP(NSQUAREDSO,NLOOPGROUPS)
      COMMON/ML5_0_FILTERS/GOODAMP,GOODHEL,HELOFFSET

      LOGICAL CHECKPHASE
      LOGICAL HELDOUBLECHECKED
      COMMON/ML5_0_INIT/CHECKPHASE, HELDOUBLECHECKED

      INTEGER SQSO_TARGET
      COMMON/ML5_0_SOCHOICE/SQSO_TARGET

      LOGICAL UVCT_REQ_SO_DONE,MP_UVCT_REQ_SO_DONE,CT_REQ_SO_DONE
     $ ,MP_CT_REQ_SO_DONE,LOOP_REQ_SO_DONE,MP_LOOP_REQ_SO_DONE
     $ ,CTCALL_REQ_SO_DONE,FILTER_SO
      COMMON/ML5_0_SO_REQS/UVCT_REQ_SO_DONE,MP_UVCT_REQ_SO_DONE
     $ ,CT_REQ_SO_DONE,MP_CT_REQ_SO_DONE,LOOP_REQ_SO_DONE
     $ ,MP_LOOP_REQ_SO_DONE,CTCALL_REQ_SO_DONE,FILTER_SO

      INTEGER I_SO
      COMMON/ML5_0_I_SO/I_SO
      INTEGER I_LIB
      COMMON/ML5_0_I_LIB/I_LIB

      COMPLEX*16 AMP(NBORNAMPS)
      COMMON/ML5_0_AMPS/AMP
      TYPE(ALOHA) W(NWAVEFUNCS)
      COMMON/ML5_0_W/W
      COMPLEX*16 WL(MAXLWFSIZE,0:LOOPMAXCOEFS-1,MAXLWFSIZE,
     $ -1:NLOOPWAVEFUNCS)
      COMPLEX*16 PL(0:3,-1:NLOOPWAVEFUNCS)
      COMMON/ML5_0_WL/WL,PL

      COMPLEX*16 AMPL(3,NCTAMPS)
      COMMON/ML5_0_AMPL/AMPL

C     
C     ----------
C     BEGIN CODE
C     ----------

C     The target squared split order contribution is already reached
C      if true.
      IF (FILTER_SO.AND.UVCT_REQ_SO_DONE) THEN
        GOTO 1001
      ENDIF

C     Amplitude(s) for UVCT diagram with ID 215
      CALL FFV1_0(W(6),W(7),W(5),GC_5,AMPL(2,263))
      AMPL(2,263)=AMPL(2,263)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 216
      CALL FFV1_0(W(6),W(7),W(5),GC_5,AMPL(1,264))
      AMPL(1,264)=AMPL(1,264)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     Amplitude(s) for UVCT diagram with ID 217
      CALL FFV1_0(W(6),W(8),W(4),GC_5,AMPL(2,265))
      AMPL(2,265)=AMPL(2,265)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 218
      CALL FFV1_0(W(6),W(8),W(4),GC_5,AMPL(1,266))
      AMPL(1,266)=AMPL(1,266)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     Amplitude(s) for UVCT diagram with ID 219
      CALL FFV1_0(W(6),W(2),W(9),GC_5,AMPL(2,267))
      AMPL(2,267)=AMPL(2,267)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 220
      CALL FFV1_0(W(6),W(2),W(9),GC_5,AMPL(1,268))
      AMPL(1,268)=AMPL(1,268)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     Amplitude(s) for UVCT diagram with ID 221
      CALL FFV1_0(W(10),W(11),W(5),GC_5,AMPL(2,269))
      AMPL(2,269)=AMPL(2,269)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 222
      CALL FFV1_0(W(10),W(11),W(5),GC_5,AMPL(1,270))
      AMPL(1,270)=AMPL(1,270)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     Amplitude(s) for UVCT diagram with ID 223
      CALL FFV2_5_0(W(10),W(8),W(3),GC_22,GC_23,AMPL(2,271))
      AMPL(2,271)=AMPL(2,271)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 224
      CALL FFV2_5_0(W(10),W(8),W(3),GC_22,GC_23,AMPL(1,272))
      AMPL(1,272)=AMPL(1,272)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     Amplitude(s) for UVCT diagram with ID 225
      CALL FFV1_0(W(12),W(11),W(4),GC_5,AMPL(2,273))
      AMPL(2,273)=AMPL(2,273)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 226
      CALL FFV1_0(W(12),W(11),W(4),GC_5,AMPL(1,274))
      AMPL(1,274)=AMPL(1,274)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     Amplitude(s) for UVCT diagram with ID 227
      CALL FFV2_5_0(W(12),W(7),W(3),GC_22,GC_23,AMPL(2,275))
      AMPL(2,275)=AMPL(2,275)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 228
      CALL FFV2_5_0(W(12),W(7),W(3),GC_22,GC_23,AMPL(1,276))
      AMPL(1,276)=AMPL(1,276)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     Amplitude(s) for UVCT diagram with ID 229
      CALL FFV1_0(W(1),W(11),W(9),GC_5,AMPL(2,277))
      AMPL(2,277)=AMPL(2,277)*(4.0D0*UVWFCT_G_1_1EPS)
C     Amplitude(s) for UVCT diagram with ID 230
      CALL FFV1_0(W(1),W(11),W(9),GC_5,AMPL(1,278))
      AMPL(1,278)=AMPL(1,278)*(2.0D0*UVWFCT_G_1+2.0D0*UVWFCT_G_2)
C     At this point, all UVCT amps needed for (QCD=6), i.e. of split
C      order ID=1, are computed.
      IF(FILTER_SO.AND.SQSO_TARGET.EQ.1) GOTO 3000

      GOTO 1001
 3000 CONTINUE
      UVCT_REQ_SO_DONE=.TRUE.
 1001 CONTINUE
      END
