class SmokeGuardTest(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.directory = Path(self.temp.name)
        names = ('MT', 'MZ', 'MW', 'MH', 'WT', 'WZ', 'WW', 'WH', 'MU')
        values = (173.3, 91.188, 80.419, 125., 0., 0., 0., 0., 1000.)
        self.words = {name: promoted_words(value) for name, value in zip(names, values)}
        self.prepared = {'smoke': {'output_contract': {'header': 'EW_CARD_SMOKE_V1', 'model_words': self.words}}}
        self.lines = ['EW_CARD_SMOKE_V1'] + ['MODEL ' + name + ' ' + ' '.join(self.words[name]) for name in names]
        self.lines += ['MODEL_IEEE_FLAGS F F F F T', 'COMPLETE_CARD_ONLY']
        for name, value in {'inputs.json': json.dumps({'after': {}}), 'ldd.txt': '',
                            'symbols.txt': '', 'stdout.log': '', 'param.log': 'model reader log\n'}.items():
            (self.directory / name).write_text(value)
        self.write()

    def write(self):
        (self.directory / 'card-smoke.hex').write_text('\n'.join(self.lines) + '\n')

    def reject(self):
        self.write()
        with self.assertRaises(ValueError):
            control.smoke_output(self.directory, self.prepared)

    def test_complete_fixture_is_accepted(self):
        result = control.smoke_output(self.directory, self.prepared)
        self.assertEqual(len(result), 6)

    def test_zero_exit_style_begin_only_is_rejected(self):
        self.lines = self.lines[:1]
        self.reject()

    def test_footer_cannot_replace_missing_model_records(self):
        self.lines = self.lines[:4] + ['COMPLETE_CARD_ONLY']
        self.reject()

    def test_single_mass_bit_change_is_rejected(self):
        row = self.lines[1].split();row[2] = f'{int(row[2], 16) ^ 1:016X}'
        self.lines[1] = ' '.join(row)
        self.reject()

    def test_single_qp_bit_change_is_rejected(self):
        row = self.lines[1].split();row[3] = f'{int(row[3], 16) ^ 1:016X}'
        self.lines[1] = ' '.join(row)
        self.reject()

    def test_model_order_is_bound(self):
        self.lines[1], self.lines[2] = self.lines[2], self.lines[1]
        self.reject()

    def test_bad_ieee_flags_are_rejected(self):
        self.lines[-2] = 'MODEL_IEEE_FLAGS F F F F UNKNOWN'
        self.reject()

    def test_empty_parameter_log_rejected(self):
        (self.directory / 'param.log').write_text('')
        self.reject()

    def test_undeclared_output_rejected(self):
        (self.directory / 'extra.txt').write_text('unexpected')
        self.reject()

    def test_extra_record_after_footer_rejected(self):
        self.lines.append('EXTRA')
        self.reject()


class SupervisorTest(unittest.TestCase):
    def test_launch_failure_is_retained_without_cleanup_of_unknown_pid(self):
        with mock.patch.object(control.shutil, 'which', return_value='/mock/taskset'), \
             mock.patch.object(control.subprocess, 'Popen', side_effect=OSError('mock launch failure')), \
             mock.patch.object(control.os, 'killpg') as kill, \
             mock.patch.object(control, 'relative', return_value='mock-private-dir'):
            event = control.bounded(['/mock/check'], Path('.'), io.StringIO(), time.monotonic() + 2, {})
        self.assertEqual(event['error']['type'], 'OSError')
        self.assertNotIn('pid', event)
        self.assertFalse(control.successful(event))
        kill.assert_not_called()

    def test_interrupted_child_is_reaped_and_event_retained(self):
        process = mock.Mock(pid=12345, returncode=-9)
        with mock.patch.object(control.shutil, 'which', return_value='/mock/taskset'), \
             mock.patch.object(control.subprocess, 'Popen', return_value=process), \
             mock.patch.object(control, 'check_interrupt', side_effect=control.ExecutionInterrupted('mock SIGTERM')), \
             mock.patch.object(control.os, 'killpg') as kill, \
             mock.patch.object(control, 'relative', return_value='mock-private-dir'):
            event = control.bounded(['/mock/check'], Path('.'), io.StringIO(), time.monotonic() + 2, {})
        self.assertEqual(event['pid'], 12345)
        self.assertEqual(event['error']['type'], 'ExecutionInterrupted')
        self.assertEqual(event['returncode'], -9)
        self.assertFalse(control.successful(event))
        kill.assert_called_once_with(12345, control.signal.SIGKILL)
        process.wait.assert_called_once_with()

    def test_expired_deadline_prevents_launch(self):
        with mock.patch.object(control.subprocess, 'Popen') as launch:
            with self.assertRaises(ValueError):
                control.bounded(['/mock/check'], Path('.'), io.StringIO(), time.monotonic() - 1, {})
        launch.assert_not_called()


if __name__ == '__main__':
    unittest.main()
